from http.server import BaseHTTPRequestHandler,HTTPServer
from pathlib import Path
from urllib.parse import parse_qs
import base64,json
root=Path('docs/evidence/t185-20261005').resolve();root.mkdir(parents=True,exist_ok=True)
allowed={'paper-detail.jpg','paper-saved.jpg','question-list.jpg','question-detail.jpg','question-editor.jpg','student-detail.jpg','student-source.jpg'}
form=b'<html lang="zh"><meta charset="utf-8"><title>T185 local capture</title><h1>T185 screenshot evidence</h1><form method="post"><label>Filename<input name="filename"></label><label>Screenshot data<textarea name="data"></textarea></label><button>Save screenshot</button></form></html>'
class Receiver(BaseHTTPRequestHandler):
    def do_GET(self):self.send_response(200);self.send_header('Content-Type','text/html; charset=utf-8');self.end_headers();self.wfile.write(form)
    def do_POST(self):
        size=int(self.headers.get('Content-Length','0'))
        if not 0<size<5000000:self.send_error(400);return
        fields=parse_qs(self.rfile.read(size).decode());name=fields.get('filename',[''])[0]
        if name not in allowed:self.send_error(400);return
        data=base64.b64decode(fields['data'][0],validate=True)
        if data[:2]!=bytes([255,216]):self.send_error(400);return
        path=(root/name).resolve();assert path.parent==root
        path.write_bytes(data)
        self.send_response(201);self.send_header('Content-Type','text/plain; charset=utf-8');self.end_headers();self.wfile.write(('Saved '+name+' '+str(len(data))+' bytes').encode())
    def log_message(self,fmt,*args):print(fmt%args,flush=True)
print('CAPTURE_READY http://127.0.0.1:8910',flush=True)
HTTPServer(('127.0.0.1',8910),Receiver).serve_forever()
